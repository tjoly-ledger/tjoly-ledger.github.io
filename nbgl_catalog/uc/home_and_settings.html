nbgl_useCaseHomeAndSettings

The app home page, with its optional action button, plus the auto-paginated settings and info pages reached through the top-right gear.

void nbgl_useCaseHomeAndSettings(const char                   *appName,
                                 const nbgl_icon_details_t    *appIcon,
                                 const char                   *tagline,
                                 const uint8_t                 initSettingPage,
                                 const nbgl_genericContents_t *settingContents,
                                 const nbgl_contentInfoList_t *infosList,
                                 const nbgl_homeAction_t      *action,
                                 nbgl_callback_t               quitCallback);
Default tagline 3 screens

A NULL tagline makes the SDK build the default one from the app name.

#define SWITCH_BLIND_SIGNING 0
#define SWITCH_NONCE         1
#define SWITCHES_NB          2

#define FIRST_SWITCH_TOKEN FIRST_USER_TOKEN

static nbgl_contentSwitch_t     switches[SWITCHES_NB];
static const char *const        INFO_TYPES[] = {"Version", "Developer"};
static const char *const        INFO_CONTENTS[] = {APPVERSION, "Ledger"};
static const nbgl_contentInfoList_t info_list = {
    .nbInfos = ARRAYLEN(INFO_TYPES),
    .infoTypes = INFO_TYPES,
    .infoContents = INFO_CONTENTS,
};

static nbgl_content_t         setting_contents[1];
static nbgl_genericContents_t setting_generic_contents;

static void settings_controls_callback(int token, uint8_t index, int page) {
    UNUSED(index);
    UNUSED(page);

    // Each switch owns a token, and NBGL redraws it from `initState` on every
    // page draw: a switch whose token is not handled here springs back as soon
    // as the page is drawn again.
    int switch_index = token - FIRST_SWITCH_TOKEN;
    if (switch_index < 0 || switch_index >= SWITCHES_NB) {
        return;
    }

    switches[switch_index].initState
        = (switches[switch_index].initState == ON_STATE) ? OFF_STATE : ON_STATE;
}

static void build_settings(void) {
    memset(switches, 0, sizeof(switches));

    switches[SWITCH_BLIND_SIGNING].text = "Blind signing";
    switches[SWITCH_BLIND_SIGNING].subText = "Allow transactions\nthat cannot be decoded";
    switches[SWITCH_BLIND_SIGNING].initState = OFF_STATE;
    switches[SWITCH_BLIND_SIGNING].token = FIRST_SWITCH_TOKEN;

    switches[SWITCH_NONCE].text = "Display nonce";
    switches[SWITCH_NONCE].subText = "Show the nonce\nin transactions";
    switches[SWITCH_NONCE].initState = ON_STATE;
    switches[SWITCH_NONCE].token = FIRST_SWITCH_TOKEN + 1;

    memset(setting_contents, 0, sizeof(setting_contents));
    setting_contents[0].type = SWITCHES_LIST;
    setting_contents[0].content.switchesList.nbSwitches = SWITCHES_NB;
    setting_contents[0].content.switchesList.switches = switches;
    setting_contents[0].contentActionCallback = settings_controls_callback;

    setting_generic_contents.callbackCallNeeded = false;
    setting_generic_contents.contentsList = setting_contents;
    setting_generic_contents.nbContents = ARRAYLEN(setting_contents);
}

// A NULL tagline makes the SDK build the default one from the app name:
// "This app enables signing transactions on the <appName> network."
nbgl_useCaseHomeAndSettings(APPNAME,
                            &CATALOG_ICON_APP,
                            NULL,
                            INIT_HOME_PAGE,
                            &setting_generic_contents,
                            &info_list,
                            NULL,
                            catalog_on_quit);
