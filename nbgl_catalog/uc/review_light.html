nbgl_operationType_t operationType

type of operation (Operation, Transaction, Message)

base type

  • TYPE_TRANSACTIONFor operations transferring a coin or taken from an account to another
  • TYPE_MESSAGEFor operations signing a message that will not be broadcast on the blockchain
  • TYPE_OPERATIONFor other types of operation (generic type)

flags, combined with |

  • SKIPPABLE_OPERATIONThis is to use in nbgl_operationType_t when the operation is skippable. This is used to add a *Skip* button on the first review page.draws “Skip”
  • BLIND_OPERATIONThis is to use in nbgl_operationType_t when the operation is "blind" This is used to indicate a warning with a top-right button in review first & last page
  • RISKY_OPERATIONThis is to use in nbgl_operationType_t when the operation is concerned by an internal warning. This is used to indicate a warning with a top-right button in review first & last page
  • NO_THREAT_OPERATIONThis is to use in nbgl_operationType_t when the operation is concerned by an internal information. This is used to indicate an info with a top-right button in review first & last page
  • ADDRESS_BOOK_OPERATIONThis is to use in nbgl_operationType_t when the operation is related to "Address Book" to adapt wordings.

const nbgl_contentTagValueList_t * tagValueList

list of tag/value pairs

This structure contains a list of [tag,value] pairs

  • pairs const nbgl_contentTagValue_t *array of [tag,value] pairs (nbPairs items). If NULL, callback is used instead
  • callback nbgl_contentTagValueCallback_tfunction to call to retrieve a given pair
  • nbPairs uint8_tnumber of pairs in pairs array (or max number of pairs to retrieve with callback)
  • startIndex uint8_tindex of the first pair to get with callback
  • hideEndOfLastLine boolif set to true, replace 3 last chars of last line by "..."
  • nbMaxLinesForValue uint8_tif > 0, set the max number of lines for value field.
  • token uint8_tthe token that will be used as argument of the callback if icon in any tag/value pair is touched (index is the index of the pair in pairs[])
  • smallCaseForValue boolif set to true, a 24px font is used for value text, otherwise a 32px font is used
  • wrapping boolif set to true, value text will be wrapped on ' ' to avoid cutting words
  • actionCallback nbgl_contentActionCallback_tcalled when a valueIcon is touched on a given pair

#define MAX_PAIRS 4

static nbgl_contentTagValue_t     pairs[MAX_PAIRS];
static nbgl_contentTagValueList_t pairList;

static nbgl_reviewStatusType_t signed_status;
static nbgl_reviewStatusType_t rejected_status;

/** Fill the pair list with a plain coin transfer, used by most variants. */
static void fill_transfer_pairs(void) {
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "To";
    pairs[1].value = "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b";
    pairs[2].item = "Fees";
    pairs[2].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 3;
    pairList.wrapping = true;
}

/** The status page an app shows once the user has answered the review. */
static void uc_review_choice(bool confirm) {
    nbgl_useCaseReviewStatus(confirm ? signed_status : rejected_status, catalog_on_quit);
}

// Same flow as nbgl_useCaseReview(), but the last page validates with a
// simple button instead of a long press: for operations with no stake.
fill_transfer_pairs();
signed_status = STATUS_TYPE_OPERATION_SIGNED;
rejected_status = STATUS_TYPE_OPERATION_REJECTED;

nbgl_useCaseReviewLight(TYPE_OPERATION,
                        &pairList,
                        &CATALOG_ICON_APP,
                        "Review operation",
                        NULL,
                        "Confirm operation?",
                        uc_review_choice);
