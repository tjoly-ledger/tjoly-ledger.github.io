static const char *const PAGE_TITLES[] = {"First page", "Second page", "Third page"};
static const char *const PAGE_TEXTS[] = {
    "This use case lets the app fill each page on demand.",
    "The navigation bar and the back arrow come for free.",
    "Return false from the callback to end the navigation.",
};

static bool uc_navigable_content_nav(uint8_t page, nbgl_pageContent_t *content) {
    // The callback is called every time the user navigates: fill `content`
    // with the page to draw, and return false once there is no more page.
    if (page >= ARRAYLEN(PAGE_TITLES)) {
        return false;
    }

    content->type = CENTERED_INFO;
    content->centeredInfo.style = LARGE_CASE_INFO;
    content->centeredInfo.text1 = PAGE_TITLES[page];
    content->centeredInfo.text2 = PAGE_TEXTS[page];

    return true;
}

static void uc_navigable_content_three_pages(void) {
    nbgl_useCaseNavigableContent("Help",
                                 0,
                                 ARRAYLEN(PAGE_TITLES),
                                 catalog_on_quit,
                                 uc_navigable_content_nav,
                                 NULL);
}
