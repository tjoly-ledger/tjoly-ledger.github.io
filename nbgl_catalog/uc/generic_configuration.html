static nbgl_content_t         config_contents[2];
static nbgl_genericContents_t config_generic_contents;


static const char *const NETWORK_NAMES[] = {"Mainnet", "Testnet", "Devnet"};
static const char *const BAR_TEXTS[] = {"Derivation path", "Custom node"};
static const uint8_t     BAR_TOKENS[] = {FIRST_USER_TOKEN + 10, FIRST_USER_TOKEN + 11};

// A titled set of configuration pages, built from any mix of content
// types: here a radio choice list, then a list of touchable bars.
memset(config_contents, 0, sizeof(config_contents));

config_contents[0].type = CHOICES_LIST;
config_contents[0].content.choicesList.names = NETWORK_NAMES;
config_contents[0].content.choicesList.nbChoices = ARRAYLEN(NETWORK_NAMES);
config_contents[0].content.choicesList.initChoice = 0;
config_contents[0].content.choicesList.token = FIRST_USER_TOKEN;

config_contents[1].type = BARS_LIST;
config_contents[1].content.barsList.barTexts = BAR_TEXTS;
config_contents[1].content.barsList.tokens = BAR_TOKENS;
config_contents[1].content.barsList.nbBars = ARRAYLEN(BAR_TEXTS);

config_generic_contents.callbackCallNeeded = false;
config_generic_contents.contentsList = config_contents;
config_generic_contents.nbContents = ARRAYLEN(config_contents);

nbgl_useCaseGenericConfiguration(
    "Network", 0, &config_generic_contents, catalog_on_quit);
