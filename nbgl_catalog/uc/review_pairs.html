nbgl_useCaseReview

The layout fields a tag/value pair carries itself, which decide how a review paginates and draws it. The list-level ones are not here: nbgl_useCaseReview() overrides smallCaseForValue, nbMaxLinesForValue and hideEndOfLastLine on every page it draws, so an app cannot set them.

void nbgl_useCaseReview(nbgl_operationType_t              operationType,
                        const nbgl_contentTagValueList_t *tagValueList,
                        const nbgl_icon_details_t        *icon,
                        const char                       *reviewTitle,
                        const char                       *reviewSubTitle,
                        const char                       *finishTitle,
                        nbgl_choiceCallback_t             choiceCallback);
forcePageStart 5 screens

Breaks the page before this pair, to keep a group of pairs together on a page of its own.

#define RECIPIENT "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b"

static nbgl_contentTagValue_t     pairs[3];
static nbgl_contentTagValueList_t pairList;

/** What the page behind the chevron of an alias is built from. */
static nbgl_contentValueExt_t extension;

/** The status page shown once the user has answered the review. */
static void uc_pairs_choice(bool confirm) {
    nbgl_useCaseReviewStatus(
        confirm ? STATUS_TYPE_TRANSACTION_SIGNED : STATUS_TYPE_TRANSACTION_REJECTED,
        catalog_on_quit);
}

/** The review itself is the ordinary one: only the pairs carry the alias. */
static void uc_pairs_review(void) {
    nbgl_useCaseReview(TYPE_TRANSACTION,
                       &pairList,
                       &CATALOG_ICON_APP,
                       "Review transaction\nto send BOL",
                       NULL,
                       "Sign transaction\nto send BOL?",
                       uc_pairs_choice);
}

static void uc_review_pairs_force_page_start(void) {
    // The three pairs below fit on one page on every product. `forcePageStart`
    // on the third one breaks the page before it, so the fees start a page of
    // their own: that is how a group of pairs is kept together.
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "To";
    pairs[1].value = "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b";
    pairs[2].item = "Fees";
    pairs[2].value = "0.00012 BOL";
    pairs[2].forcePageStart = 1;

    pairList.pairs = pairs;
    pairList.nbPairs = 3;
    pairList.wrapping = true;

    uc_pairs_review();
}

static void uc_review_pairs_centered_info(void) {
    // `centeredInfo` turns a pair into a page of its own, drawn like a review
    // cover: the icon of `valueIcon`, `item` as the title and `value` as the
    // description, under which the SDK adds its own "Swipe to review".
    // `valueIcon` shares its union with `extension`, so a pair cannot be both
    // a centered info and an alias.
    pairs[0].item = "Send 1.2345 BOL";
    pairs[0].value = "to Alice";
    pairs[0].centeredInfo = 1;
    pairs[0].valueIcon = &CATALOG_ICON_APP;
    pairs[1].item = "Fees";
    pairs[1].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 2;
    pairList.wrapping = true;

    uc_pairs_review();
}

A value too long for a page 6 screens

Not a field but what NBGL does without one: it builds a TAG_VALUE_DETAILS page with a More button, the one content type a use case cannot be passed.

static void uc_review_pairs_long_value(void) {
    // A value too long for one page is not truncated by the app: NBGL turns
    // that page into a TAG_VALUE_DETAILS content of its own, with a "More"
    // button opening the full value. TAG_VALUE_DETAILS is the one content
    // type a use case cannot be passed, the SDK only ever builds it here.
    pairs[0].item = "Raw data";
    pairs[0].value
        = "0x9f0a1b2c3d4e5f60718293a4b5c6d7e8f90a1b2c3d4e5f60718293a4b5c6d7e8"
          "f90a1b2c3d4e5f60718293a4b5c6d7e8f90a1b2c3d4e5f60718293a4b5c6d7e8f9"
          "0a1b2c3d4e5f60718293a4b5c6d7e8f90a1b2c3d4e5f60718293a4b5c6d7e8f90a"
          "1b2c3d4e5f60718293a4b5c6d7e8f90a1b2c3d4e5f60718293a4b5c6d7e8f9";

    pairList.pairs = pairs;
    pairList.nbPairs = 1;

    uc_pairs_review();
}
