nbgl_useCaseAdvancedReview

The configurable review: a bitfield of predefined warnings drives the intro pages, and the top-right button of the first and last pages opens the matching details.

Needs ENABLE_NBGL_QRCODE = 1 in the app Makefile.

The transaction cannot be decoded.

static nbgl_contentTagValue_t     pairs[MAX_PAIRS];
static nbgl_contentTagValueList_t pairList;

static nbgl_reviewStatusType_t signed_status;
static nbgl_reviewStatusType_t rejected_status;

/** Fill the pair list with a plain coin transfer, used by most variants. */
static void fill_transfer_pairs(void) {
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "To";
    pairs[1].value = "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b";
    pairs[2].item = "Fees";
    pairs[2].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 3;
    pairList.wrapping = true;
}

/** The status page an app shows once the user has answered the review. */
static void uc_review_choice(bool confirm) {
    nbgl_useCaseReviewStatus(confirm ? signed_status : rejected_status, catalog_on_quit);
}


static void uc_advanced_review_blind_signing(void) {
    // The transaction cannot be decoded.
    // The BLIND_OPERATION flag of `operationType` is what puts the top-right
    // button on the review pages; the bit of `predefinedSet` is what builds
    // the intro page and the details behind that button.
    // This warning quotes no provider, so dAppProvider / reportProvider /
    // reportUrl are left unset: see the Web3 Checks variants for those.
    fill_transfer_pairs();
    signed_status = STATUS_TYPE_TRANSACTION_SIGNED;
    rejected_status = STATUS_TYPE_TRANSACTION_REJECTED;
    warning.predefinedSet = 1 << BLIND_SIGNING_WARN;

    nbgl_useCaseAdvancedReview(TYPE_TRANSACTION | BLIND_OPERATION,
                               &pairList,
                               &CATALOG_ICON_APP,
                               "Review transaction\nto send BOL",
                               NULL,
                               "Sign transaction\nto send BOL?",
                               NULL,
                               &warning,
                               uc_review_choice);
}

No warning, custom tip-box 6 screens

With an empty predefinedSet there is no warning page, and the tip-box the app provides is the one displayed at the bottom of the first page; touching it opens its modal. This is the only way to a custom tip-box, since a predefined warning tip takes priority over the caller's.

static nbgl_tipBox_t tip_box;

static const char *const TIP_INFO_TYPES[] = {"Raw data"};
static const char *const TIP_INFO_CONTENTS[] = {"0x9f0a1b2c3d4e5f60"};

static void uc_advanced_review_tip_box(void) {
    // With an empty predefinedSet there is no predefined warning, so the
    // tip-box the app provides is the one displayed: touching it opens a modal
    // built from these infos. That is the only path to a custom tip-box, as
    // initWarningTipBox() (nbgl_use_case.c:2875) gives a predefined warning
    // tip priority over the caller's.
    fill_transfer_pairs();
    signed_status = STATUS_TYPE_TRANSACTION_SIGNED;
    rejected_status = STATUS_TYPE_TRANSACTION_REJECTED;

    tip_box.text = "Raw transaction data";
    tip_box.icon = &SMALL_INFO_CIRCLE_ICON;
    tip_box.modalTitle = "Raw transaction data";
    tip_box.type = INFOS_LIST;
    tip_box.infos.nbInfos = ARRAYLEN(TIP_INFO_TYPES);
    tip_box.infos.infoTypes = TIP_INFO_TYPES;
    tip_box.infos.infoContents = TIP_INFO_CONTENTS;

    warning.predefinedSet = 0;

    nbgl_useCaseAdvancedReview(TYPE_TRANSACTION,
                               &pairList,
                               &CATALOG_ICON_APP,
                               "Review transaction\nto send BOL",
                               NULL,
                               "Sign transaction\nto send BOL?",
                               &tip_box,
                               &warning,
                               uc_review_choice);
}
