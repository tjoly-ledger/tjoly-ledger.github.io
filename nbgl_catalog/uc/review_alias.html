nbgl_useCaseReview

A tag/value pair whose value is an alias: the review shows the short name with a chevron, and the page behind it justifies the alias. The aliasType of nbgl_contentValueExt_t picks that page.

Needs ENABLE_NBGL_QRCODE = 1 in the app Makefile. 

void nbgl_useCaseReview(nbgl_operationType_t              operationType,
                        const nbgl_contentTagValueList_t *tagValueList,
                        const nbgl_icon_details_t        *icon,
                        const char                       *reviewTitle,
                        const char                       *reviewSubTitle,
                        const char                       *finishTitle,
                        nbgl_choiceCallback_t             choiceCallback);
ENS_ALIAS 6 screens

A domain name resolved by the Ledger backend, which the SDK says itself under the full address.

#define RECIPIENT "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b"

static nbgl_contentTagValue_t     pairs[3];
static nbgl_contentTagValueList_t pairList;

/** What the page behind the chevron of an alias is built from. */
static nbgl_contentValueExt_t extension;

/** The status page shown once the user has answered the review. */
static void uc_pairs_choice(bool confirm) {
    nbgl_useCaseReviewStatus(
        confirm ? STATUS_TYPE_TRANSACTION_SIGNED : STATUS_TYPE_TRANSACTION_REJECTED,
        catalog_on_quit);
}

/**
 * A plain transfer whose recipient is an alias: `aliasValue` is what puts the
 * chevron next to the value, and `extension` what the page behind it shows.
 * Every variant below only changes that extension.
 */
static void fill_alias_pairs(const char *alias) {
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "To";
    pairs[1].value = alias;
    pairs[1].aliasValue = 1;
    pairs[1].extension = &extension;
    pairs[2].item = "Fees";
    pairs[2].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 3;
    pairList.wrapping = true;
}

/** The review itself is the ordinary one: only the pairs carry the alias. */
static void uc_pairs_review(void) {
    nbgl_useCaseReview(TYPE_TRANSACTION,
                       &pairList,
                       &CATALOG_ICON_APP,
                       "Review transaction\nto send BOL",
                       NULL,
                       "Sign transaction\nto send BOL?",
                       uc_pairs_choice);
}

static void uc_review_alias_ens(void) {
    // ENS_ALIAS: the page repeats the alias as its title, then the full
    // address, under a sentence the SDK writes itself ("ENS names are
    // resolved by Ledger backend."), so `explanation` is not used here.
    fill_alias_pairs("vitalik.eth");

    extension.aliasType = ENS_ALIAS;
    extension.fullValue = RECIPIENT;

    uc_pairs_review();
}

static nbgl_contentInfoList_t alias_infos;

static void uc_review_alias_info_list(void) {
    // INFO_LIST_ALIAS: the page behind the chevron is a list of infos rather
    // than a single value, for an alias that needs several lines to be
    // justified. `backText` titles that page, instead of the tag name.
    fill_alias_pairs("Boilerplate contract");

    alias_infos.nbInfos = ARRAYLEN(ALIAS_INFO_TYPES);
    alias_infos.infoTypes = ALIAS_INFO_TYPES;
    alias_infos.infoContents = ALIAS_INFO_CONTENTS;

    extension.aliasType = INFO_LIST_ALIAS;
    extension.backText = "Contract details";
    extension.infolist = &alias_infos;

    uc_pairs_review();
}

static void uc_review_alias_tag_value_list(void) {
    // TAG_VALUE_LIST_ALIAS: the page behind the chevron is itself a list of
    // tag/value pairs. Like the main list, it must live in RAM.
    // `backText` is left unset because this type ignores it: the SDK works
    // out the modal title, then calls displayTagValueListModal() without it
    // (nbgl_use_case.c:1451). The page comes up with no title.
    fill_alias_pairs("Swap 1.2345 BOL");

    alias_pairs[0].item = "Pool";
    alias_pairs[0].value = "BOL / USDC";
    alias_pairs[1].item = "Minimum received";
    alias_pairs[1].value = "4 521.08 USDC";

    alias_pair_list.pairs = alias_pairs;
    alias_pair_list.nbPairs = 2;
    alias_pair_list.wrapping = true;

    extension.aliasType = TAG_VALUE_LIST_ALIAS;
    extension.tagValuelist = &alias_pair_list;

    uc_pairs_review();
}
