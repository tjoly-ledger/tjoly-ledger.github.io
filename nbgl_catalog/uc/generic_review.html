nbgl_useCaseGenericReview

Any mix of content types paginated as a review, with a reject footer, when the predefined reviews do not fit.

void nbgl_useCaseGenericReview(const nbgl_genericContents_t *contents,
                               const char                   *rejectText,

TAG_VALUE_LIST + INFO_LONG_PRESS 3 screens

Pairs, then a centered info validated by a long press.

#define MAX_PAIRS 4

static nbgl_contentTagValue_t     pairs[MAX_PAIRS];
static nbgl_contentTagValueList_t pairList;

static nbgl_reviewStatusType_t signed_status;
static nbgl_reviewStatusType_t rejected_status;

/** Fill the pair list with a plain coin transfer, used by most variants. */
static void fill_transfer_pairs(void) {
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "To";
    pairs[1].value = "0x4A2b1F8c3D7e5A9b0C1d2E3f4A5b6C7d8E9f0A1b";
    pairs[2].item = "Fees";
    pairs[2].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 3;
    pairList.wrapping = true;
}

#define GENERIC_LONG_PRESS_TOKEN (FIRST_USER_TOKEN + 0)
#define GENERIC_BUTTON_TOKEN     (FIRST_USER_TOKEN + 1)
#define GENERIC_CONFIRM_TOKEN    (FIRST_USER_TOKEN + 2)
#define GENERIC_TIP_BOX_TOKEN    (FIRST_USER_TOKEN + 3)

static nbgl_content_t         generic_contents[2];
static nbgl_genericContents_t generic_review_contents;

/**
 * Unlike the predefined reviews, a generic review has no choice callback: what
 * the user does with a content is reported here, through the token that
 * content carries.
 */
static void uc_generic_review_action(int token, uint8_t index, int page) {
    UNUSED(index);
    UNUSED(page);

    switch (token) {
        case GENERIC_LONG_PRESS_TOKEN:
        case GENERIC_BUTTON_TOKEN:
        case GENERIC_CONFIRM_TOKEN:
            nbgl_useCaseReviewStatus(STATUS_TYPE_TRANSACTION_SIGNED, catalog_on_quit);
            break;
        default:
            // A tip-box is the app's own: NBGL only reports the tap, so a
            // content it does not act on simply leaves the page as it is.
            break;
    }
}

static void uc_generic_review_plain(void) {
    // Any mix of content types paginated as a review, with a reject footer.
    // Here a tag/value page, then a centered info with a long press button.
    fill_transfer_pairs();
    memset(generic_contents, 0, sizeof(generic_contents));

    generic_contents[0].type = TAG_VALUE_LIST;
    generic_contents[0].content.tagValueList = pairList;

    generic_contents[1].type = INFO_LONG_PRESS;
    generic_contents[1].content.infoLongPress.text = "Sign transaction\nto send BOL?";
    generic_contents[1].content.infoLongPress.icon = &CATALOG_ICON_APP;
    generic_contents[1].content.infoLongPress.longPressText = "Hold to sign";
    generic_contents[1].content.infoLongPress.longPressToken = GENERIC_LONG_PRESS_TOKEN;
    generic_contents[1].contentActionCallback = uc_generic_review_action;

    generic_review_contents.callbackCallNeeded = false;
    generic_review_contents.contentsList = generic_contents;
    generic_review_contents.nbContents = ARRAYLEN(generic_contents);

    nbgl_useCaseGenericReview(&generic_review_contents, "Reject", catalog_on_quit);
}

static void uc_generic_review_tag_value_confirm(void) {
    // TAG_VALUE_CONFIRM puts the pairs, a black button and a footer on the
    // same page, so the review is answered without a long press. NBGL only
    // draws the button on the last page of the list: a list that does not fit
    // on one page is shown as ordinary tag/value pages until that last one.
    // These two pairs fit on one page on every product.
    pairs[0].item = "Amount";
    pairs[0].value = "1.2345 BOL";
    pairs[1].item = "Fees";
    pairs[1].value = "0.00012 BOL";

    pairList.pairs = pairs;
    pairList.nbPairs = 2;
    pairList.wrapping = true;

    memset(generic_contents, 0, sizeof(generic_contents));

    generic_contents[0].type = TAG_VALUE_CONFIRM;
    generic_contents[0].content.tagValueConfirm.tagValueList = pairList;
    generic_contents[0].content.tagValueConfirm.confirmationText = "Sign transaction";
    generic_contents[0].content.tagValueConfirm.confirmationToken = GENERIC_CONFIRM_TOKEN;
    // cancelText / cancelToken are left unset on purpose: a generic content
    // page draws them from its navigation, so the footer is the rejectText
    // given to nbgl_useCaseGenericReview() and those two fields are never
    // read (nbgl_page.c:134).
    generic_contents[0].contentActionCallback = uc_generic_review_action;

    generic_review_contents.callbackCallNeeded = false;
    generic_review_contents.contentsList = generic_contents;
    generic_review_contents.nbContents = 1;

    nbgl_useCaseGenericReview(&generic_review_contents, "Reject", catalog_on_quit);
}

    generic_contents[0].content.extendedCenter.contentCenter.icon = &CATALOG_ICON_APP;

        = "Review transaction\nto send BOL";

        = "You will be asked to sign it on the last page.";

    generic_contents[0].content.extendedCenter.tipBox.icon = &SMALL_INFO_CIRCLE_ICON;
    generic_contents[0].content.extendedCenter.tipBox.token = GENERIC_TIP_BOX_TOKEN;
    generic_contents[0].contentActionCallback = uc_generic_review_action;

    generic_contents[1].type = INFO_LONG_PRESS;
    generic_contents[1].content.infoLongPress.text = "Sign transaction\nto send BOL?";
