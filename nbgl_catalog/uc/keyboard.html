nbgl_useCaseKeyboard

A text entry page, with either a confirmation button or dynamic suggestion buttons.

Needs ENABLE_NBGL_KEYBOARD = 1 in the app Makefile. 

void nbgl_useCaseKeyboard(const nbgl_keyboardParams_t *params, nbgl_callback_t backCallback);
With a confirmation button 1 screen 
#define ENTRY_MAX_LEN 32

static nbgl_keyboardParams_t keyboard_params;

/** The app owns the buffer the keyboard writes into. */
static char entry_buffer[ENTRY_MAX_LEN + 1];

// A confirmation button sits under the entry area; the keyboard starts in
// letters mode and `casing` decides how the first letter is capitalised.
keyboard_params.type = KEYBOARD_WITH_BUTTON;
keyboard_params.title = "Enter the account name";
keyboard_params.entryBuffer = entry_buffer;
keyboard_params.entryMaxLen = ENTRY_MAX_LEN;
keyboard_params.mode = MODE_LETTERS;
keyboard_params.lettersOnly = false;
keyboard_params.casing = LOCKED_UPPER_CASE;
keyboard_params.confirmationParams.buttonText = "Confirm";
keyboard_params.confirmationParams.onButtonCallback = uc_keyboard_on_button;

nbgl_useCaseKeyboard(&keyboard_params, catalog_on_quit);

#define ENTRY_MAX_LEN 32

static nbgl_keyboardParams_t keyboard_params;

/** The app owns the buffer the keyboard writes into. */
static char entry_buffer[ENTRY_MAX_LEN + 1];

// Suggestion buttons replace the confirmation button; the update callback
// is called on every key press, to refresh them and to grey out the keys
// that cannot lead to a word.
suggestions[0] = "abandon";
suggestions[1] = "ability";
suggestions[2] = "able";
suggestions[3] = "about";

keyboard_params.type = KEYBOARD_WITH_SUGGESTIONS;
keyboard_params.title = "Enter word no. 1";
keyboard_params.entryBuffer = entry_buffer;
keyboard_params.entryMaxLen = ENTRY_MAX_LEN;
keyboard_params.mode = MODE_LETTERS;
keyboard_params.lettersOnly = true;
keyboard_params.numbered = true;
keyboard_params.number = 1;
keyboard_params.casing = LOWER_CASE;
keyboard_params.suggestionParams.buttons = suggestions;
keyboard_params.suggestionParams.firstButtonToken = FIRST_USER_TOKEN;
keyboard_params.suggestionParams.onButtonCallback = uc_keyboard_on_suggestion;
keyboard_params.suggestionParams.updateButtonsCallback = uc_keyboard_update_suggestions;

nbgl_useCaseKeyboard(&keyboard_params, catalog_on_quit);
